# tests/bench/CMakeLists.txt
#
# Microbenchmark suite + regression gate.
#
# Each `bench_*.cpp` is its own executable that consumes the nanobench
# single-header library. Bench targets are built unconditionally with
# the rest of the test tree, but their ctest registrations carry the
# `BENCH` label so the default fast suite (`ctest -LE "SLOW|LOAD|BENCH"`)
# skips them. The `formulon_bench_check` ctest entry runs every bench
# in turn, captures their JSON outputs into `${CMAKE_BINARY_DIR}/bench_results/`,
# then invokes `tools/bench/check_regression.py` against the committed
# baseline. A regression of more than 20% wall-clock fails the gate.
#
# nanobench depends on `<exception>` / `<system_error>` internally; the
# bench TUs opt out of `-fno-exceptions` via a target-local compile
# option. Production code is unaffected.

include(${CMAKE_SOURCE_DIR}/cmake/FetchNanobench.cmake)

set(_BENCH_NAMES
  bench_recalc_1m_chain
  bench_recalc_1m_wide
  bench_io_100mb_read
  bench_eval_hot_path
  bench_compile_to_vm
)

set(_BENCH_RESULTS_DIR "${CMAKE_BINARY_DIR}/bench_results")
file(MAKE_DIRECTORY "${_BENCH_RESULTS_DIR}")

foreach(_bench IN LISTS _BENCH_NAMES)
  add_executable(${_bench} ${_bench}.cpp)
  target_link_libraries(${_bench}
    PRIVATE
      formulon_static
      formulon::nanobench
  )
  target_include_directories(${_bench}
    PRIVATE
      ${CMAKE_SOURCE_DIR}/src
  )
  # Bench TUs use nanobench's exception-throwing internals. Override the
  # project-wide `-fno-exceptions` for these targets only. We deliberately
  # do NOT call `formulon_apply_warnings` because that re-applies the
  # restricted flag set; we want loose flags here so nanobench's own
  # macro expansions (which include `try` / `catch`) compile.
  #
  # We deliberately do NOT define `NDEBUG` differently from the rest of
  # the build: the bench TUs include the same headers as `formulon_core`,
  # and any `#ifdef NDEBUG` branch in those headers must agree on the
  # struct layout used across translation units. Mismatched NDEBUG state
  # between the bench TU and the linked library produces use-after-free
  # bugs at exit time when destructors disagree on member layout. The
  # baseline is therefore measured against whatever build mode the
  # caller's `build/` directory is using; numbers from a Debug build
  # will be slower than from a Release build, but they are internally
  # consistent and compare cleanly against a baseline measured in the
  # same configuration.
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU|AppleClang")
    target_compile_options(${_bench}
      PRIVATE
        -fexceptions
        -frtti
        -Wno-conversion
        -Wno-shadow
    )
  endif()

  set_target_properties(${_bench} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests/bench"
  )
endforeach()

# `bench_all` umbrella target — builds (and, if invoked as a build
# target, depends on) every bench executable. Convenient for
# `cmake --build build --target bench_all`.
add_custom_target(bench_all DEPENDS ${_BENCH_NAMES})

# Driver script: runs every bench with a JSON output path under
# `${CMAKE_BINARY_DIR}/bench_results/<name>.json`. Used by both the
# `BENCH`-labelled ctest entry and humans invoking the binaries
# directly.
set(_BENCH_RUNNER "${CMAKE_CURRENT_BINARY_DIR}/run_bench_all.sh")
file(WRITE "${_BENCH_RUNNER}.in"
"#!/bin/sh
# Generated by tests/bench/CMakeLists.txt. Do not edit by hand.
set -eu
BENCH_DIR='@CMAKE_BINARY_DIR@/tests/bench'
RESULTS_DIR='@_BENCH_RESULTS_DIR@'
mkdir -p \"$RESULTS_DIR\"
")
foreach(_bench IN LISTS _BENCH_NAMES)
  file(APPEND "${_BENCH_RUNNER}.in"
"echo \"== $BENCH_DIR/${_bench} ==\"
\"$BENCH_DIR/${_bench}\" --json \"$RESULTS_DIR/${_bench}.json\"
")
endforeach()
file(APPEND "${_BENCH_RUNNER}.in"
"echo 'bench_all: wrote results to '\"$RESULTS_DIR\"
")

configure_file("${_BENCH_RUNNER}.in" "${_BENCH_RUNNER}" @ONLY)
file(CHMOD "${_BENCH_RUNNER}"
  PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)

# Bench-only ctest entry. The driver runs every bench (capturing JSON),
# then hands the results directory to the regression checker. Labelled
# `BENCH` so the default fast suite skips it.
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(
    NAME formulon_bench_check
    COMMAND ${CMAKE_COMMAND} -E env
      "FORMULON_BENCH_RUNNER=${_BENCH_RUNNER}"
      "FORMULON_BENCH_RESULTS=${_BENCH_RESULTS_DIR}"
      "FORMULON_BENCH_BASELINE=${CMAKE_SOURCE_DIR}/tools/bench/baseline.json"
      ${Python3_EXECUTABLE}
      ${CMAKE_SOURCE_DIR}/tools/bench/check_regression.py
        --runner "${_BENCH_RUNNER}"
        --results "${_BENCH_RESULTS_DIR}"
        --baseline "${CMAKE_SOURCE_DIR}/tools/bench/baseline.json"
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
  )
  set_tests_properties(formulon_bench_check
    PROPERTIES
      LABELS "BENCH"
      RUN_SERIAL TRUE
      TIMEOUT 600
  )
endif()
